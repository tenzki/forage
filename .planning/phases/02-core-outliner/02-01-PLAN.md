   pub async fn move_node(
       state: State<'_, AppState>,
       id: String,
       new_parent_id: Option<String>,
       new_position: String,
   ) -> Result<Node, AppError> {
       // Verify node exists
       let _ = get_node(State::clone(&state), id.clone()).await?;
       let now = Utc::now().to_rfc3339();

       sqlx::query("UPDATE nodes SET parent_id = ?1, position = ?2, updated_at = ?3 WHERE id = ?4")
           .bind(&new_parent_id)
           .bind(&new_position)
           .bind(&now)
           .bind(&id)
           .execute(&state.db)
           .await?;

       get_node(state, id).await
   }
